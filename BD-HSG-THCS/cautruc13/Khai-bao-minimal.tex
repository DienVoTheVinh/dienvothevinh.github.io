% Chapter — Thiết kế bất đối xứng hiện đại kiểu Nhật: Số hiệu lớn làm mờ nền + Hanko stamp
\titleformat{\chapter}[display]
  {\normalfont\sffamily}
  {\vspace{-1.5cm}%
   \begin{flushright}
     \begin{tikzpicture}
       \node[anchor=east, font=\sffamily\fontsize{72pt}{72pt}\selectfont\bfseries\color{vmAccent!25}] at (0,0) {\ifnum\value{chapter}<10 0\fi\arabic{chapter}};
       \node[anchor=east, font=\sffamily\fontsize{11pt}{12pt}\selectfont\bfseries\color{vmAccent}] at (-0.2,-0.9) {CHUYÊN ĐỀ};
     \end{tikzpicture}
   \end{flushright}}
  {-1.2cm}
  {\begin{flushleft}
     \hanko\enspace{\Huge\bfseries\color{vmAccent}#1}\\[15pt]
     \noindent{\color{vmAccent}\hrule height 2.5pt}\vskip 2.5pt{\color{vmOk}\hrule height 1.8pt width 4.5cm}
   \end{flushleft}}

% Section — Thẻ số hiệu màu đỏ son nổi bật + chữ đỏ son đồng bộ, gạch chân ngắn tinh tế
\titleformat{\section}
  {\normalfont\Large\sffamily\bfseries\color{vmAccent}}
  {}
  {0pt}
  {\begin{tikzpicture}[baseline=0pt]
     \node[anchor=base, fill=vmAccent, inner sep=3.5pt, rounded corners=2pt] (A) {\color{white}\sffamily\bfseries\Large\strut\thesection};
   \end{tikzpicture}\enspace#1}[\vspace{3pt}\noindent\textcolor{vmLine}{\rule{3.5cm}{1.5pt}}]
\titlespacing*{\section}{0pt}{3.5ex plus 1ex minus .2ex}{2.5ex plus .2ex}

% Subsection — Thẻ số hiệu Vàng đặc, tiêu đề xanh Navy
\titleformat{\subsection}
  {\normalfont\large\sffamily\bfseries\color{vmAccent}}
  {}
  {0pt}
  {\begin{tikzpicture}[baseline=0pt]
     \node[anchor=base, fill=vmOk, inner sep=2.5pt, rounded corners=1.5pt] (B) {\color{white}\sffamily\bfseries\large\strut\thesubsection};
   \end{tikzpicture}\enspace#1}
\titlespacing*{\subsection}{0pt}{2.8ex}{1.5ex}

% Subsubsection — Thẻ số hiệu Đỏ đặc, tiêu đề Charcoal/Đen

  {\normalfont\normalsize\sffamily\bfseries\color{vmInk}}
  {}
  {0pt}
  {\begin{tikzpicture}[baseline=0pt]
     \node[anchor=base, fill=vmWarn, inner sep=2.2pt, rounded corners=1.2pt] (C) {\color{white}\sffamily\bfseries\normalsize\strut\thesubsubsection};
   \end{tikzpicture}\enspace#1}
