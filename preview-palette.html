# Định nghĩa đạo hàm tại một điểm
def derivative(f, x, h=1e-5):
    dy = f(x + h) - f(x)
    dx = h
    return dy / dx

# Áp dụng cho hàm bậc hai y = x² tại x = 3
y = lambda x: x**2
slope = derivative(y, 3) # Kết quả: ~ 6.00001
